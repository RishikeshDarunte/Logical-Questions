Q.1) Print unique unsorted array Accept data in an array having duplicate value. You need
to print unique array using only 2 loop .
Input 3, 5, 3, 7, 9, 7, 3, 1 output 3, 5, 7, 9, 1


import java.util.Arrays;
public class UniqueUnsortedArray {

    public static int[] getUniqueArray(int[] arr) {  
        int[] temp = new int[arr.length]; 
        int uniqueCount = 0; 
        for (int i = 0; i < arr.length; i++) {
            boolean isDuplicate = false;
            for (int j = 0; j < uniqueCount; j++) {
                if (arr[i] == temp[j]) {
                    isDuplicate = true; 
                    break;
                    }
                 }
            if (!isDuplicate) {
                temp[uniqueCount] = arr[i]; 
                uniqueCount++; 
            }
        }

        return Arrays.copyOf(temp, uniqueCount);
    }

    public static void main(String[] args) {
        int[] arr = {3, 5, 3, 7, 9, 7, 3, 1}; 
        int[] uniqueArr = getUniqueArray(arr);

        System.out.println(Arrays.toString(uniqueArr));
    }
}

Q.2) Rearrange an array in maximum minimum form using Two Pointer Technique
Given a sorted array of positive integers, rearrange the array alternately i.e first element
should be a maximum value, at second position minimum value, at third position second
max, at fourth position second min, and so on.
Examples:
Input: arr[] = {1, 2, 3, 4, 5, 6, 7}
Output: arr[] = {7, 1, 6, 2, 5, 3, 4}



import java.util.Arrays;

public class MinMaxRearrange {

    static void rearrange(long[] arr, int n)
    {
      
        long[] temp = new long[n];

        int small = 0, large = n - 1;

            boolean flag = true;
      
        for (int i = 0; i < n; i++) {
            if (flag) {
                temp[i] = arr[large--];	 
            }
            else {
                temp[i] = arr[small++];

            }											
            flag = !flag;
        }			
	
        for (int i = 0; i < n; i++)
            arr[i] = temp[i];
    }


    public static void main(String[] args)
    {
        long[] arr = { 1, 2, 3, 4, 5, 6 };
        int n = arr.length;

        System.out.println("Original Array");
        for (int i = 0; i < n; i++)
            System.out.print(arr[i] + " ");
      
        rearrange(arr, n);

        System.out.println("\nModified Array");
        for (int i = 0; i < n; i++)
            System.out.print(arr[i] + " ");
    }
}
